   137→# ----------------------------------------------------------------------
   138→# Conexión
   139→# ----------------------------------------------------------------------
   140→def get_connection() -> sqlite3.Connection:
   141→    """Abre una conexión SQLite con filas tipo diccionario y claves foráneas activas."""
   142→    conexion = sqlite3.connect(DB_PATH)
   143→    conexion.row_factory = sqlite3.Row
   144→    conexion.execute("PRAGMA foreign_keys = ON;")
   145→    return conexion
   146→
   147→
   148→@contextmanager
   149→def _conexion() -> Iterator[sqlite3.Connection]:
   150→    """Contexto transaccional: commit al salir, rollback ante error, siempre cierra."""
   151→    con = get_connection()
   152→    try:
   153→        yield con
   154→        con.commit()
   155→    except Exception:
   156→        con.rollback()
   157→        raise
   158→    finally:
   159→        con.close()
   160→
   161→
   162→def _como_json(valor: Any) -> str:
   163→    """Serializa a texto JSON legible (tildes conservadas)."""
   164→    return json.dumps(valor, ensure_ascii=False)
   165→
   166→
   167→def _desde_json(texto: str | None, por_defecto: Any) -> Any:
   168→    """Deserializa JSON almacenado; devuelve `por_defecto` si está vacío/inválido."""
   169→    if not texto:
   170→        return por_defecto
   171→    try:
   172→        return json.loads(texto)
   173→    except (TypeError, ValueError):
   174→        return por_defecto

   182→    with _conexion() as con:
   183→        con.executescript(ESQUEMA)
   184→    log.info("Base de datos inicializada en %s", DB_PATH)

   191→    """Registra un deportista activo y devuelve su id."""

   223→def listar_deportistas(solo_activos: bool = True, categoria: str = "", texto: str = "") -> list[sqlite3.Row]:
   224→    """Lista deportistas con filtros opcionales de categoría y búsqueda por nombre."""
   225→    condiciones = ["1=1"]
   226→    parametros: list[Any] = []
   227→    if solo_activos:
   228→        condiciones.append("activo=1")
   229→    if categoria:
   230→        condiciones.append("categoria=?")
   231→        parametros.append(categoria)
   232→    if texto:
   233→        condiciones.append("LOWER(nombre) LIKE ?")
   234→        parametros.append(f"%{texto.lower()}%")
   235→    sql = "SELECT * FROM deportistas WHERE " + " AND ".join(condiciones) + " ORDER BY nombre"
   236→    with _conexion() as con:
   237→        return list(con.execute(sql, parametros).fetchall())

   257→# Registros mensuales — AT-F3
   258→# ----------------------------------------------------------------------
   259→def upsert_registro(
   260→    deportista_id: int,
   261→    periodo: str,
   262→    *,
   263→    objetivos: str = "",
   264→    pct_asistencia: float | None = None,
   265→    pct_cumplimiento: float | None = None,
   266→    resultados: str = "",
   267→    horario: list[str] | None = None,
   268→    evaluacion: str = "",
   269→    entrenador: str = "",
   270→) -> int:
   271→    """Crea o actualiza el registro AT-F3 de (deportista, periodo). Devuelve el id."""
   272→    parametros = (
   273→        deportista_id, periodo, objetivos, pct_asistencia, pct_cumplimiento,
   274→        resultados, _como_json(horario or []), evaluacion, entrenador,
   275→    )
   276→    with _conexion() as con:
   277→        cursor = con.execute(
   278→            """
   279→            INSERT INTO registros_mensuales
   280→                (deportista_id, periodo, objetivos, pct_asistencia, pct_cumplimiento,
   281→                 resultados, horario_json, evaluacion, entrenador)
   282→            VALUES (?,?,?,?,?,?,?,?,?)
   283→            ON CONFLICT(deportista_id, periodo) DO UPDATE SET
   284→                objetivos        = excluded.objetivos,
   285→                pct_asistencia   = COALESCE(excluded.pct_asistencia, registros_mensuales.pct_asistencia),
   286→                pct_cumplimiento = COALESCE(excluded.pct_cumplimiento, registros_mensuales.pct_cumplimiento),
   287→                resultados       = excluded.resultados,
   288→                horario_json     = excluded.horario_json,
   289→                evaluacion       = excluded.evaluacion,
   290→                entrenador       = excluded.entrenador,
   291→                actualizado_en   = datetime('now','localtime')
   292→            """,
   293→            parametros,
   294→        )
   295→        fila = con.execute(
   296→            "SELECT id FROM registros_mensuales WHERE deportista_id=? AND periodo=?",
   297→            (deportista_id, periodo),
   298→        ).fetchone()
   299→        log.info("Registro AT-F3 upsert: deportista=%s periodo=%s", deportista_id, periodo)
   300→        return int(fila["id"])

   303→def obtener_registro(registro_id: int) -> sqlite3.Row | None:
   304→    """Registro AT-F3 con datos del deportista (JOIN)."""
   305→    with _conexion() as con:
   306→        return con.execute(
   307→            """
   308→            SELECT r.*, d.nombre, d.deporte, d.categoria, d.sexo,
   309→                   d.car AS car_dep, d.region AS region_dep, d.entrenador AS entrenador_dep
   310→            FROM registros_mensuales r
   311→            JOIN deportistas d ON d.id = r.deportista_id
   312→            WHERE r.id=?
   313→            """,
   314→            (registro_id,),
   315→        ).fetchone()

   318→def listar_registros(periodo: str = "", deportista_id: int | None = None) -> list[sqlite3.Row]:
   319→    """Lista registros AT-F3 (opcionalmente de un periodo o deportista)."""
   320→    condiciones, parametros = ["1=1"], []
   321→    if periodo:
   322→        condiciones.append("r.periodo=?")
   323→        parametros.append(periodo)
   324→    if deportista_id is not None:
   325→        condiciones.append("r.deportista_id=?")
   326→        parametros.append(deportista_id)
   327→    sql = (
   328→        "SELECT r.*, d.nombre, d.categoria, d.deporte FROM registros_mensuales r "
   329→        "JOIN deportistas d ON d.id=r.deportista_id WHERE "
   330→        + " AND ".join(condiciones)
   331→        + " ORDER BY r.periodo DESC, d.nombre"
   332→    )
   333→    with _conexion() as con:
   334→        return list(con.execute(sql, parametros).fetchall())

   337→def agregar_pruebas_sisdena(deportista_id: int, periodo: str, nuevas: list[dict[str, Any]]) -> int:
   338→    """Fusiona participaciones SISDENA (sin duplicar) en el registro del periodo."""
   339→    clave = lambda p: (str(p.get("fecha", "")), str(p.get("competencia", "")), str(p.get("prueba", "")))
   340→    with _conexion() as con:
   341→        fila = con.execute(
   342→            "SELECT id, pruebas_sisdena_json FROM registros_mensuales WHERE deportista_id=? AND periodo=?",
   343→            (deportista_id, periodo),
   344→        ).fetchone()
   345→        actuales: list[dict[str, Any]] = _desde_json(fila["pruebas_sisdena_json"] if fila else None, [])
   346→        existentes = {clave(p) for p in actuales}
   347→        for prueba in nuevas:
   348→            if clave(prueba) not in existentes:
   349→                actuales.append(prueba)
   350→                existentes.add(clave(prueba))
   351→        if fila:
   352→            con.execute(
   353→                "UPDATE registros_mensuales SET pruebas_sisdena_json=?, actualizado_en=datetime('now','localtime') "
   354→                "WHERE id=?",
   355→                (_como_json(actuales), fila["id"]),
   356→            )
   357→        else:
   358→            con.execute(
   359→                "INSERT INTO registros_mensuales (deportista_id, periodo, pruebas_sisdena_json) VALUES (?,?,?)",
   360→                (deportista_id, periodo, _como_json(actuales)),
   361→            )
   362→        return len(actuales)

   368→def upsert_consolidado(car: str, periodo: str, datos: dict[str, Any]) -> int:
   369→    """Crea o actualiza el consolidado AT-F1 de (CAR, periodo). Devuelve el id."""
   370→    with _conexion() as con:

   372→            """
   373→            INSERT INTO consolidados_at_f1

   376→            VALUES (:car, :periodo, :matriculas_json, :eval_medica, :eval_nutricion,
   377→                    :eval_car, :observaciones, :jut_nombre)
   378→            ON CONFLICT(car, periodo) DO UPDATE SET
   379→                matriculas_json = excluded.matriculas_json,
   380→                eval_medica     = excluded.eval_medica,
   381→                eval_nutricion  = excluded.eval_nutricion,
   382→                eval_car        = excluded.eval_car,
   383→                observaciones   = excluded.observaciones,
   384→                jut_nombre      = excluded.jut_nombre,
   385→                actualizado_en  = datetime('now','localtime')
   386→            """,
   387→            {
   388→                "car": car, "periodo": periodo,
   389→                "matriculas_json": _como_json(datos.get("matriculas", {"altas": {}, "bajas": {}})),
   390→                "eval_medica": datos.get("eval_medica", ""),
   391→                "eval_nutricion": datos.get("eval_nutricion", ""),
   392→                "eval_car": datos.get("eval_car", ""),
   393→                "observaciones": datos.get("observaciones", ""),
   394→                "jut_nombre": datos.get("jut_nombre", ""),
   395→            },
   396→        )
   397→        fila = con.execute(
   398→            "SELECT id FROM consolidados_at_f1 WHERE car=? AND periodo=?", (car, periodo)

   400→        return int(fila["id"])

   431→def promedios_periodo(car: str, periodo: str) -> dict[str, Any]:
   432→    """Promedios de % asistencia / cumplimiento (AT-F3) para AT-F9."""
   433→    with _conexion() as con:
   434→        fila = con.execute(
   435→            """
   436→            SELECT AVG(r.pct_asistencia) AS asistencia, AVG(r.pct_cumplimiento) AS cumplimiento,
   437→                   COUNT(*) AS n
   438→            FROM registros_mensuales r
   439→            JOIN deportistas d ON d.id = r.deportista_id
   440→            WHERE r.periodo=? AND (?='' OR LOWER(TRIM(d.car))=LOWER(TRIM(?)))
   441→            """,
   442→            (periodo, car.strip(), car.strip()),
   443→        ).fetchone()
   444→    return {
   445→        "pct_asistencia": fila["asistencia"],
   446→        "pct_cumplimiento": fila["cumplimiento"],
   447→        "n": int(fila["n"] or 0),
   448→    }

   467→        )

   563→# Firmas (SHA-256 · solo desactivación · auditoría completa)
   564→# ----------------------------------------------------------------------
   565→def registrar_firma(rol: str, nombre: str, contenido_png: bytes) -> int:
   566→    """Guarda el PNG de la rúbrica, su hash SHA-256 y deja constancia en auditoría."""
   567→    FIRMAS_DIR.mkdir(parents=True, exist_ok=True)
   568→    hash_sha = hashlib.sha256(contenido_png).hexdigest()
   569→    archivo = FIRMAS_DIR / f"{rol}_{datetime.now():%Y%m%d_%H%M%S}.png"
   570→    archivo.write_bytes(contenido_png)
   571→    with _conexion() as con:
   572→        cursor = con.execute(
   573→            "INSERT INTO firmas (rol, nombre, archivo, sha256, activo) VALUES (?,?,?,?,1)",
   574→            (rol, nombre, str(archivo), hash_sha),
   575→        )
   576→        firma_id = int(cursor.lastrowid or 0)
   577→        con.execute("INSERT INTO asignaciones_firma (firma_id, ambito) VALUES (?, '*')", (firma_id,))
   578→        con.execute(
   579→            "INSERT INTO auditoria_firmas (firma_id, accion, detalle, hash_sha256) "
   580→            "VALUES (?, 'registro', ?, ?)",
   581→            (firma_id, f"Rol={rol} · Titular={nombre}", hash_sha),
   582→        )
   583→    log.info("Firma registrada: id=%s rol=%s sha256=%s", firma_id, rol, hash_sha[:12])
   584→    return firma_id
