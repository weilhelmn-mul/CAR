    33→  { key: 'fechaNacimiento', etiqueta: 'F. nacimiento', porDefecto: false },
    34→  { key: 'categoria', etiqueta: 'Categoría', porDefecto: true },
    35→  { key: 'sexo', etiqueta: 'Sexo', porDefecto: true },

    53→export default function Deportistas({ depIdFicha, setDepIdFicha }: { depIdFicha: number | null; setDepIdFicha: (v: number | null) => void }) {
    54→  const { lista, cargando, recargar } = useDeportistas()
    55→  const [q, setQ] = useState('')
    56→  const [fCategoria, setFCategoria] = useState('todas')
    57→  const [fSexo, setFSexo] = useState('todos')
    58→  const [fModalidad, setFModalidad] = useState('todas')
    59→  // Filtros avanzados
    60→  const [verFiltros, setVerFiltros] = useState(false)
    61→  const [fDeporte, setFDeporte] = useState('todos')
    62→  const [fNivel, setFNivel] = useState('todos')
    63→  const [fPad, setFPad] = useState('todos') // todos | si | no
    64→  const [fAval, setFAval] = useState('todos') // todos | VIGENTE | POR VENCER | EXPIRADO | SIN REGISTRO
    65→  const [fEstado, setFEstado] = useState('todos') // todos | activos | retirados
    66→  const [edadMin, setEdadMin] = useState('')
    67→  const [edadMax, setEdadMax] = useState('')
    68→  // Ordenamiento
    69→  const [orden, setOrden] = useState<{ key: string; dir: 'asc' | 'desc' } | null>(null)
    70→  // Columnas visibles (elección del usuario, persistente)
    71→  const [visibles, setVisibles] = useState<Set<ColKey>>(new Set(COLUMNAS.filter((c) => c.porDefecto).map((c) => c.key)))
    72→  useEffect(() => {
    73→    try {
    74→      const raw = localStorage.getItem(LS_COLUMNAS)
    75→      if (raw) {
    76→        const arr = JSON.parse(raw) as ColKey[]
    77→        setVisibles(new Set(COLUMNAS.filter((c) => arr.includes(c.key)).map((c) => c.key)))
    78→      }
    79→    } catch { /* usa valores por defecto */ }
    80→  }, [])
    81→  const toggleColumna = (k: ColKey, activo: boolean) => {
    82→    setVisibles((prev) => {
    83→      const nx = new Set(prev)
    84→      if (activo) nx.add(k); else nx.delete(k)
    85→      try { localStorage.setItem(LS_COLUMNAS, JSON.stringify([...nx])) } catch { /* sin persistencia */ }
    86→      return nx
    87→    })
    88→  }
    89→  const restaurarColumnas = () => {
    90→    const base = new Set(COLUMNAS.filter((c) => c.porDefecto).map((c) => c.key))
    91→    setVisibles(base)
    92→    try { localStorage.setItem(LS_COLUMNAS, JSON.stringify([...base])) } catch { /* sin persistencia */ }
    93→  }
    94→
    95→  const [dialogo, setDialogo] = useState(false)
    96→  const [editando, setEditando] = useState<Deportista | null>(null)
    97→  const [form, setForm] = useState<FormDep>(vacio)
    98→  const [guardando, setGuardando] = useState(false)
    99→
   100→  const cambiarOrden = (key: string) =>
   101→    setOrden((o) => (o?.key === key ? (o.dir === 'asc' ? { key, dir: 'desc' } : null) : { key, dir: 'asc' }))
   102→
   103→  const valorOrden = (d: Deportista, key: string): string | number => {
   104→    switch (key) {
   105→      case 'nombres': return nombreCompletoUI(d).toLowerCase()
   106→      case 'dni': return d.dni
   107→      case 'edad': return calcularEdad(d.fechaNacimiento) ?? -1
   108→      case 'fechaNacimiento': return d.fechaNacimiento?.slice(0, 10) ?? ''
   109→      case 'categoria': return d.categoria
   110→      case 'sexo': return d.sexo
   111→      case 'deporte': return d.deporte
   112→      case 'nivelTecnico': return d.nivelTecnico
   113→      case 'modalidad': return d.modalidad
   114→      case 'ingreso': return d.fechaIngreso?.slice(0, 10) ?? ''
   115→      case 'continuidad': return d.fechaVencimientoContinuidad?.slice(0, 10) ?? ''
   116→      case 'aval': return d.fechaCaducidadAval?.slice(0, 10) ?? ''
   117→      case 'pad': return d.pad ? 1 : 0
   118→      case 'procedencia': return d.procedencia
   119→      case 'telefono': return d.telefono ?? ''
   120→      case 'entrenador': return d.entrenador ?? ''
   121→      case 'estado': return d.retirado ? 1 : 0
   122→      default: return ''
   123→    }
   124→  }
   125→
   126→  const filtrados = useMemo(() => {
   127→    const t = q.trim().toLowerCase()
   128→    const min = edadMin === '' ? null : Number(edadMin)
   129→    const max = edadMax === '' ? null : Number(edadMax)
   130→    let f = lista.filter((d) => {
   131→      if (t && !`${d.apellidoPaterno} ${d.apellidoMaterno} ${d.nombres} ${d.dni}`.toLowerCase().includes(t)) return false
   132→      if (fCategoria !== 'todas' && d.categoria !== fCategoria) return false
   133→      if (fSexo !== 'todos' && d.sexo !== fSexo) return false
   134→      if (fModalidad !== 'todas' && d.modalidad !== fModalidad) return false
   135→      if (fDeporte !== 'todos' && d.deporte !== fDeporte) return false
   136→      if (fNivel !== 'todos' && d.nivelTecnico !== fNivel) return false
   137→      if (fPad === 'si' && !d.pad) return false
   138→      if (fPad === 'no' && d.pad) return false
   139→      if (fAval !== 'todos' && estadoAval(d.fechaCaducidadAval).texto !== fAval) return false
   140→      if (fEstado === 'activos' && d.retirado) return false
   141→      if (fEstado === 'retirados' && !d.retirado) return false
   142→      const e = calcularEdad(d.fechaNacimiento)
   143→      if (min !== null && !isNaN(min) && (e === null || e < min)) return false
   144→      if (max !== null && !isNaN(max) && (e === null || e > max)) return false
   145→      return true
   146→    })
   147→    if (orden) {
   148→      const { key, dir } = orden
   149→      const s = dir === 'asc' ? 1 : -1
   150→      f = [...f].sort((a, b) => {
   151→        const va = valorOrden(a, key), vb = valorOrden(b, key)
   152→        const vacioA = va === '' || va === -1, vacioB = vb === '' || vb === -1
   153→        if (vacioA && vacioB) return 0
   154→        if (vacioA) return 1 // vacíos al final siempre
   155→        if (vacioB) return -1
   156→        if (va < vb) return -1 * s
   157→        if (va > vb) return 1 * s
   158→        return 0
   159→      })
   160→    }
   161→    return f
   162→    // eslint-disable-next-line react-hooks/exhaustive-deps
   163→  }, [lista, q, fCategoria, fSexo, fModalidad, fDeporte, fNivel, fPad, fAval, fEstado, edadMin, edadMax, orden])
   164→
   165→  const numFiltros = [q, fCategoria, fSexo, fModalidad, fDeporte, fNivel, fPad, fAval, fEstado, edadMin, edadMax]
   166→    .filter((v) => v !== '' && v !== 'todos' && v !== 'todas').length
   167→
   168→  const limpiarFiltros = () => {
   169→    setQ(''); setFCategoria('todas'); setFSexo('todos'); setFModalidad('todas')
   170→    setFDeporte('todos'); setFNivel('todos'); setFPad('todos'); setFAval('todos'); setFEstado('todos')
   171→    setEdadMin(''); setEdadMax(''); setOrden(null)
   172→  }
   173→
   174→  const abrirNuevo = () => { setEditando(null); setForm(vacio); setDialogo(true) }
   175→  const abrirEdicion = (d: Deportista) => {
   176→    setEditando(d)
   177→    setForm({
   178→      apellidoPaterno: d.apellidoPaterno, apellidoMaterno: d.apellidoMaterno, nombres: d.nombres, dni: d.dni,
   179→      sexo: d.sexo, fechaNacimiento: d.fechaNacimiento?.slice(0, 10) ?? '', fechaIngreso: d.fechaIngreso?.slice(0, 10) ?? '',
   180→      deporte: d.deporte, prueba: d.prueba ?? '', nivelTecnico: d.nivelTecnico, modalidad: d.modalidad,
   181→      categoria: d.categoria, telefono: d.telefono ?? '', procedencia: d.procedencia, pad: d.pad,
   182→      entrenador: d.entrenador ?? '', activo: d.activo, retirado: d.retirado,
   183→      fechaVencimientoContinuidad: d.fechaVencimientoContinuidad?.slice(0, 10) ?? '',
   184→      fechaCaducidadAval: d.fechaCaducidadAval?.slice(0, 10) ?? '',
   185→    })
   186→    setDialogo(true)
   187→  }
   188→
   189→  const guardar = async () => {
   190→    if (!form.apellidoPaterno || !form.nombres || !form.dni || !form.fechaNacimiento) {
   191→      toast({ title: 'Datos incompletos', description: 'Apellidos, nombres, DNI y fecha de nacimiento son obligatorios.', variant: 'destructive' })
   192→      return
   193→    }
   194→    setGuardando(true)
   195→    try {
   196→      if (editando) {
   197→        await api(`/api/deportistas/${editando.id}`, { method: 'PUT', body: JSON.stringify(form) })
   198→        toast({ title: 'Deportista actualizado' })
   199→      } else {
   200→        await api('/api/deportistas', { method: 'POST', body: JSON.stringify(form) })
   201→        toast({ title: 'Deportista registrado' })
   202→      }
   203→      setDialogo(false)
   204→      recargar()
   205→    } catch (e) {
   206→      toast({ title: 'Error', description: e instanceof Error ? e.message : 'Error', variant: 'destructive' })
   207→    } finally { setGuardando(false) }
   208→  }
   209→
   210→  const eliminar = async (d: Deportista) => {
   211→    if (!confirm(`¿Eliminar a ${nombreCompletoUI(d)}? Se borrarán también sus resultados.`)) return
   212→    try {
   213→      await api(`/api/deportistas/${d.id}`, { method: 'DELETE' })
   214→      toast({ title: 'Deportista eliminado' })
   215→      recargar()
   216→    } catch (e) {
   217→      toast({ title: 'Error', description: e instanceof Error ? e.message : 'Error', variant: 'destructive' })
   218→    }
   219→  }
   220→
   221→  const set = (k: keyof FormDep, v: string | boolean) => setForm((f) => ({ ...f, [k]: v }))
   222→
   223→  const visiblesList = COLUMNAS.filter((c) => visibles.has(c.key))
   224→  const colSpan = 3 + visiblesList.length
   225→  const thOrden = (key: string, label: string, className = 'text-red-900') => (
   226→    <TableHead className={`px-2 ${className}`}>
   227→      <button type="button" className="inline-flex items-center gap-1 font-medium hover:underline" onClick={() => cambiarOrden(key)}>
   228→        {label}
   229→        {orden?.key === key
   230→          ? (orden.dir === 'asc' ? <ArrowUp className="h-3 w-3" /> : <ArrowDown className="h-3 w-3" />)
   231→          : <ArrowUpDown className="h-3 w-3 opacity-40" />}
   232→      </button>
   233→    </TableHead>
   234→  )
   235→  const celda = (d: Deportista, key: ColKey) => {
   236→    switch (key) {
   237→      case 'dni': return <span className="text-sm">{d.dni}</span>
   238→      case 'edad': {
   239→        const e = calcularEdad(d.fechaNacimiento)
   240→        return <span className="text-sm font-semibold text-slate-700">{e !== null ? e : '—'}</span>
   241→      }
   242→      case 'fechaNacimiento': return <span className="text-sm">{fmtFecha(d.fechaNacimiento) || '—'}</span>
   243→      case 'categoria': return <BadgeCat>{d.categoria}</BadgeCat>
   244→      case 'sexo': return <span className="text-sm">{d.sexo === 'FEMENINO' ? 'F' : 'M'}</span>
   245→      case 'deporte': return <span className="text-sm">{d.deporte}<span className="block text-xs text-slate-500">{d.prueba ?? '—'}</span></span>
   246→      case 'nivelTecnico': return <span className="text-sm">{d.nivelTecnico}</span>
   247→      case 'modalidad': return <span className="text-sm">{d.modalidad}</span>
   248→      case 'ingreso': return <span className="text-sm">{fmtFecha(d.fechaIngreso) || '—'}</span>
   249→      case 'continuidad': return (
   250→        <span className={`text-sm ${d.fechaVencimientoContinuidad ? 'text-slate-700' : 'text-slate-400'}`}>{fmtFecha(d.fechaVencimientoContinuidad) || '—'}</span>
   251→      )
   252→      case 'aval': return (
   253→        <span>
   254→          <span className={`inline-flex items-center rounded px-2 py-0.5 text-xs font-bold ${estadoAval(d.fechaCaducidadAval).clase}`}>
   255→            {estadoAval(d.fechaCaducidadAval).texto}
   256→          </span>
   257→          {d.fechaCaducidadAval && <span className="block text-xs text-slate-500">{fmtFecha(d.fechaCaducidadAval)}</span>}
   258→        </span>
   259→      )
   260→      case 'pad': return <span className={`text-sm font-semibold ${d.pad ? 'text-green-700' : 'text-slate-400'}`}>{d.pad ? 'SI' : 'NO'}</span>
   261→      case 'procedencia': return <span className="text-sm">{d.procedencia}</span>
   262→      case 'telefono': return <span className="text-sm">{d.telefono ?? '—'}</span>
   263→      case 'entrenador': return <span className="text-sm text-slate-600">{d.entrenador ?? '—'}</span>
   264→      case 'estado': return d.retirado
   265→        ? <span className="text-xs font-bold text-slate-500">RETIRADO</span>
   266→        : <span className="text-xs font-bold text-green-700">ACTIVO</span>
   267→      default: return null
   268→    }
   269→  }
   270→  const edadForm = calcularEdad(form.fechaNacimiento)
   271→
   272→  return (
   273→    <div className="space-y-4">
   274→      <div className="flex flex-wrap items-center gap-2">
   275→        <div className="relative min-w-52 flex-1">
   276→          <UserSearch className="absolute left-2.5 top-2.5 h-4 w-4 text-slate-400" />
   277→          <Input placeholder="Buscar por apellido o DNI…" className="pl-8" value={q} onChange={(e) => setQ(e.target.value)} />
   278→        </div>
   279→        <Select value={fCategoria} onValueChange={setFCategoria}>
   280→          <SelectTrigger className="w-28"><SelectValue /></SelectTrigger>
   281→          <SelectContent>{['todas', ...CATEGORIAS].map((c) => <SelectItem key={c} value={c}>{c.toUpperCase()}</SelectItem>)}</SelectContent>
   282→        </Select>
   283→        <Select value={fSexo} onValueChange={setFSexo}>
   284→          <SelectTrigger className="w-28"><SelectValue /></SelectTrigger>
   285→          <SelectContent>{['todos', ...SEXOS].map((c) => <SelectItem key={c} value={c}>{c.toUpperCase()}</SelectItem>)}</SelectContent>
   286→        </Select>
   287→        <Select value={fModalidad} onValueChange={setFModalidad}>
   288→          <SelectTrigger className="w-32"><SelectValue /></SelectTrigger>
   289→          <SelectContent>{['todas', ...MODALIDADES].map((c) => <SelectItem key={c} value={c}>{c}</SelectItem>)}</SelectContent>
   290→        </Select>
   291→        <Button variant={verFiltros ? 'default' : 'outline'} className={verFiltros ? 'bg-red-800 hover:bg-red-900' : 'border-red-800 text-red-900 hover:bg-red-50'} onClick={() => setVerFiltros((v) => !v)}>
   292→          <Filter className="h-4 w-4" /> Filtros
   293→          {numFiltros > 0 && <span className="ml-1 inline-flex h-5 min-w-5 items-center justify-center rounded-full bg-red-800 px-1 text-[11px] font-bold text-white">{numFiltros}</span>}
   294→        </Button>
   295→        <Popover>
   296→          <PopoverTrigger asChild>
   297→            <Button variant="outline" className="border-red-800 text-red-900 hover:bg-red-50">
   298→              <Columns3 className="h-4 w-4" /> Columnas
   299→              <span className="ml-1 text-xs text-slate-500">{visiblesList.length}/{COLUMNAS.length}</span>
   300→            </Button>
   301→          </PopoverTrigger>
   302→          <PopoverContent className="w-64" align="end">
   303→            <div className="mb-2 flex items-center justify-between">
   304→              <p className="text-xs font-bold uppercase tracking-wide text-slate-600">Variables a visualizar</p>
   305→              <button type="button" className="text-xs font-medium text-red-800 hover:underline" onClick={restaurarColumnas}>Restaurar</button>
   306→            </div>
   307→            <div className="grid max-h-72 gap-1 overflow-y-auto pr-1">
   308→              {COLUMNAS.map((c) => (
   309→                <label key={c.key} className="flex cursor-pointer items-center gap-2 rounded px-1 py-0.5 text-sm text-slate-700 hover:bg-slate-50">
   310→                  <Checkbox checked={visibles.has(c.key)} onCheckedChange={(v) => toggleColumna(c.key, v === true)} className="accent-red-800" />
   311→                  {c.etiqueta}
   312→                </label>
   313→              ))}
   314→            </div>
   315→          </PopoverContent>
   316→        </Popover>
   317→        <Button className="bg-red-800 hover:bg-red-900" onClick={abrirNuevo}><Plus className="h-4 w-4" /> Nuevo</Button>
   318→      </div>
   319→
   320→      {verFiltros && (
   321→        <div className="rounded-xl border bg-white p-3 shadow-sm">
   322→          <div className="grid gap-2 sm:grid-cols-2 lg:grid-cols-4">
   323→            <label className="grid gap-1 text-xs font-medium text-slate-600">Deporte
   324→              <Select value={fDeporte} onValueChange={setFDeporte}>
   325→                <SelectTrigger><SelectValue /></SelectTrigger>
   326→                <SelectContent>{['todos', ...DEPORTES].map((c) => <SelectItem key={c} value={c}>{c === 'todos' ? 'TODOS' : c}</SelectItem>)}</SelectContent>
   327→              </Select>
   328→            </label>
   329→            <label className="grid gap-1 text-xs font-medium text-slate-600">Nivel técnico
   330→              <Select value={fNivel} onValueChange={setFNivel}>
   331→                <SelectTrigger><SelectValue /></SelectTrigger>
   332→                <SelectContent>{['todos', ...NIVELES].map((c) => <SelectItem key={c} value={c}>{c.toUpperCase()}</SelectItem>)}</SelectContent>
   333→              </Select>
   334→            </label>
   335→            <label className="grid gap-1 text-xs font-medium text-slate-600">Seguro PAD
   336→              <Select value={fPad} onValueChange={setFPad}>
   337→                <SelectTrigger><SelectValue /></SelectTrigger>
   338→                <SelectContent>
   339→                  <SelectItem value="todos">TODOS</SelectItem>
   340→                  <SelectItem value="si">CON PAD</SelectItem>
   341→                  <SelectItem value="no">SIN PAD</SelectItem>
   342→                </SelectContent>
   343→              </Select>
   344→            </label>
   345→            <label className="grid gap-1 text-xs font-medium text-slate-600">Aval médico
   346→              <Select value={fAval} onValueChange={setFAval}>
   347→                <SelectTrigger><SelectValue /></SelectTrigger>
   348→                <SelectContent>
   349→                  {['todos', 'VIGENTE', 'POR VENCER', 'EXPIRADO', 'SIN REGISTRO'].map((c) => <SelectItem key={c} value={c}>{c.toUpperCase()}</SelectItem>)}
